Structured Text (ST) Reference

// ============================================================
// PLC_PRG — Automatic Bottle Filling Plant
// ============================================================

PROGRAM PLC_PRG
VAR
    M_SystemRunning     : BOOL;
    M_BottleInPosition  : BOOL;
    M_FillComplete      : BOOL;
    TON_Fill            : TON;
    CTU_BottleCount     : CTU;
END_VAR

// Network 1: Start/Stop seal-in
M_SystemRunning := (I_Start_PB OR M_SystemRunning)
                    AND I_Stop_PB
                    AND I_EStop_OK;

// Network 2: Conveyor runs unless a bottle is at the station
Q_ConveyorMotor := M_SystemRunning AND NOT M_BottleInPosition;

// Network 3: Latch bottle-in-position on photo-eye
IF I_Bottle_Position_Sensor THEN
    M_BottleInPosition := TRUE;
END_IF

// Network 4: Fill duration timer
TON_Fill(IN := M_BottleInPosition, PT := T#5s);
M_FillComplete := TON_Fill.Q;

// Network 5: Fill valve output
Q_FillValve := M_BottleInPosition AND NOT M_FillComplete;

// Networks 6 & 7: Clear flags once the filled bottle departs
IF NOT I_Bottle_Position_Sensor AND M_FillComplete THEN
    M_BottleInPosition := FALSE;
    M_FillComplete      := FALSE;
END_IF

// Network 8: Batch counter
CTU_BottleCount(CU := M_FillComplete, R := I_Reset_Count_PB, PV := 50);
Q_BatchCompleteLamp := CTU_BottleCount.Q;
