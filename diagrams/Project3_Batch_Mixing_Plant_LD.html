Structured Text (ST) Reference

// ============================================================
// PLC_PRG — Batch Mixing Plant
// ============================================================

PROGRAM PLC_PRG
VAR
    M_SystemRunning  : BOOL;
    M_TankFull       : BOOL;
    M_MixDone        : BOOL;
    TON_Mix          : TON;
    CTU_BatchCount   : CTU;
END_VAR

// Network 1: Start/Stop seal-in
M_SystemRunning := (I_Start_PB OR M_SystemRunning)
                    AND I_Stop_PB
                    AND I_EStop_OK;

// Network 2: Inlet valve fills the vessel until tank-full latch sets
Q_InletValve := M_SystemRunning AND NOT M_TankFull;

// Network 3: Latch tank-full on high-level sensor
IF I_Level_High_Sensor THEN
    M_TankFull := TRUE;
END_IF

// Network 4: Mix duration timer
TON_Mix(IN := M_TankFull, PT := T#10s);
M_MixDone := TON_Mix.Q;

// Network 5: Mixer motor runs only while mixing is in progress
Q_MixerMotor := M_TankFull AND NOT M_MixDone;

// Network 6: Drain valve opens once mixing is complete
Q_DrainValve := M_MixDone;

// Networks 7 & 8: Clear flags once the vessel is confirmed empty
IF NOT I_Level_Low_Sensor AND M_MixDone THEN
    M_TankFull := FALSE;
    M_MixDone  := FALSE;
END_IF

// Network 9: Batch counter
CTU_BatchCount(CU := M_MixDone, R := I_Reset_Count_PB, PV := 20);
Q_BatchCompleteLamp := CTU_BatchCount.Q;
