Structured Text (ST) Reference

// ============================================================
// PLC_PRG — Automated Warehouse Conveyor
// ============================================================

PROGRAM PLC_PRG
VAR
    M_SystemRunning  : BOOL;
    M_Zone2Occupied  : BOOL;
    M_JamTimerDone   : BOOL;
    M_JamAlarm       : BOOL;
    TON_JamDetect    : TON;
END_VAR

// Network 1: Start/Stop seal-in with jam-alarm interlock
M_SystemRunning := (I_Start_PB OR M_SystemRunning)
                    AND I_Stop_PB
                    AND I_EStop_OK
                    AND NOT M_JamAlarm;

// Network 2: Zone 1 accumulation control
Q_Zone1Motor := M_SystemRunning AND NOT M_Zone2Occupied;

// Network 3: Zone 2 runs continuously
Q_Zone2Motor := M_SystemRunning;

// Network 4: Latch Zone 2 occupied on entry sensor
IF I_Zone2_Entry_Sensor THEN
    M_Zone2Occupied := TRUE;
END_IF

// Network 5: Clear Zone 2 occupied on exit sensor
IF I_Zone2_Exit_Sensor THEN
    M_Zone2Occupied := FALSE;
END_IF

// Network 6: Diverter arm routes to Bay B when switch selected
Q_DiverterArm := I_Carton_At_Diverter AND I_Bay_Select_Switch;

// Network 7: Jam detection — carton held past max dwell time
TON_JamDetect(IN := I_Zone1_Entry_Sensor AND NOT I_Zone1_Exit_Sensor, PT := T#8s);
M_JamTimerDone := TON_JamDetect.Q;

// Network 8: Latch alarm flag and drive stack light together
IF M_JamTimerDone THEN
    M_JamAlarm := TRUE;
END_IF
Q_AlarmLamp := M_JamAlarm;

// Network 9: Manual jam-alarm reset
IF I_Jam_Reset_PB THEN
    M_JamAlarm := FALSE;
END_IF
