Structured Text (ST) Reference

// ============================================================
// PLC_PRG — Smart Conveyor Sorting System
// ============================================================

PROGRAM PLC_PRG
VAR
    M_SystemRunning   : BOOL;
    M_RejectFlag      : BOOL;
    M_PusherTmrDone   : BOOL;
    TON_Pusher        : TON;
END_VAR

// Network 1: Start/Stop seal-in
M_SystemRunning := (I_Start_PB OR M_SystemRunning)
                    AND I_Stop_PB
                    AND I_EStop_OK;

// Network 2: Conveyor motor
Q_ConveyorMotor := M_SystemRunning;

// Network 3: Reject classification (SET, edge-triggered by sensors)
IF I_Photo_Sensor AND I_Height_Sensor THEN
    M_RejectFlag := TRUE;
END_IF

// Network 4: Pusher travel-delay timer
TON_Pusher(IN := M_RejectFlag, PT := T#2s);
M_PusherTmrDone := TON_Pusher.Q;

// Network 5: Pusher actuation
Q_PusherSolenoid := M_PusherTmrDone;

// Network 6: Reject flag reset on pusher retract
IF I_Pusher_Retract_LS THEN
    M_RejectFlag := FALSE;
END_IF
